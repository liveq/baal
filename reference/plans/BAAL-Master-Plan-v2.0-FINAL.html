🎯 BAAL이란?

세계 최초 엔터테인먼트형 자율 규제 커뮤니티 + 리워드 광고 통합 플랫폼

  • ⚖️ 커뮤니티 법정: 억울한 제재를 역전재판으로 해결
  • 🍯 꿀단지: 광고 시청으로 실제 현금 적립
  • 📌 꼬리표: 유저의 대표 글을 커뮤니티가 선정
  • 🎭 실시간 관전: 재판을 팝콘 먹으며 관람

핵심 차별점

  • 평판(명예) ≠ 포인트(현금): 완전 분리로 공정성 확보

  • 삭제 불가 증거: 바알의 저울에서는 모든 발언 영구 기록
  • 리워드 광고 분리: AdSense 정책 완벽 준수

6.1 핵심 테이블

users (회원)

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  email VARCHAR(255) UNIQUE NOT NULL,
  oauth_provider VARCHAR(20) NOT NULL,
  nickname VARCHAR(50) NOT NULL,
  
  -- 평판 (명예, 거래 불가)
  reputation INTEGER DEFAULT 0,
  trust_level VARCHAR(20) DEFAULT 'bronze',
  
  -- 포인트 (현금, 출금 가능)
  points INTEGER DEFAULT 0,
  
  -- 통계
  posts_count INTEGER DEFAULT 0,
  comments_count INTEGER DEFAULT 0,
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  
  is_admin BOOLEAN DEFAULT FALSE,
  is_banned BOOLEAN DEFAULT FALSE,
  
  INDEX idx_email (email),
  INDEX idx_reputation (reputation DESC)
);

posts (게시글)

CREATE TABLE posts (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  board_type VARCHAR(50) NOT NULL, -- 'ai', 'court' 등
  title VARCHAR(200) NOT NULL,
  content TEXT NOT NULL,
  
  -- 바알의 저울 전용
  is_chat_mode BOOLEAN DEFAULT FALSE,
  
  -- 통계
  views INTEGER DEFAULT 0,
  upvotes INTEGER DEFAULT 0,
  downvotes INTEGER DEFAULT 0,
  comments_count INTEGER DEFAULT 0,
  
  -- 상태
  is_deleted BOOLEAN DEFAULT FALSE,
  is_best BOOLEAN DEFAULT FALSE,
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  
  INDEX idx_board_created (board_type, created_at DESC),
  INDEX idx_board_upvotes (board_type, upvotes DESC)
);

comments (댓글)

CREATE TABLE comments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  post_id UUID REFERENCES posts(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  parent_id UUID REFERENCES comments(id),

  upvotes INTEGER DEFAULT 0,
  downvotes INTEGER DEFAULT 0,
  
  -- 바알의 저울 전용
  can_edit BOOLEAN DEFAULT TRUE,
  can_delete BOOLEAN DEFAULT TRUE,
  
  is_deleted BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT NOW(),
  
  INDEX idx_post_created (post_id, created_at)
);

votes (추천/비추)

CREATE TABLE votes (

7.1 Supabase RLS (Row Level Security) 정책

users 테이블

-- 모든 사람이 읽기 가능 (프로필 조회)
CREATE POLICY "Anyone can view users"
ON users FOR SELECT
USING (true);

-- 본인만 수정 가능
CREATE POLICY "Users can update own profile"
ON users FOR UPDATE
USING (auth.uid() = id);

posts 테이블

-- 모든 사람이 읽기 가능
CREATE POLICY "Anyone can view posts"
ON posts FOR SELECT
USING (is_deleted = false);

-- 로그인한 사람만 작성 가능
CREATE POLICY "Authenticated users can insert posts"
ON posts FOR INSERT
WITH CHECK (auth.uid() = user_id);

-- 본인만 수정/삭제 가능
CREATE POLICY "Users can update own posts"
ON posts FOR UPDATE
USING (auth.uid() = user_id);

CREATE POLICY "Users can delete own posts"
ON posts FOR DELETE
USING (auth.uid() = user_id);

comments 테이블

-- 모든 사람이 읽기 가능
CREATE POLICY "Anyone can view comments"
ON comments FOR SELECT
USING (is_deleted = false);

-- 로그인한 사람만 작성
CREATE POLICY "Authenticated users can insert comments"
ON comments FOR INSERT
WITH CHECK (auth.uid() = user_id);

-- 본인만 수정/삭제 (단, can_edit/can_delete = true인 경우만)
CREATE POLICY "Users can update own comments if allowed"
ON comments FOR UPDATE
USING (auth.uid() = user_id AND can_edit = true);

CREATE POLICY "Users can delete own comments if allowed"
ON comments FOR DELETE
USING (auth.uid() = user_id AND can_delete = true);

구현 (Vercel Edge Config)

// middleware.ts
import { Ratelimit } from '@upstash/ratelimit'
import { Redis } from '@upstash/redis'

const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(10, '1 h'), // 10 requests per hour
})

export async function middleware(request: Request) {
  const ip = request.headers.get('x-forwarded-for')
  const { success } = await ratelimit.limit(ip)
  
  if (!success) {
    return new Response('Too many requests', { status: 429 })
  }
}

7.4 민감 정보 암호화

// 출금 계좌 정보 암호화
import crypto from 'crypto'

const ENCRYPTION_KEY = process.env.ENCRYPTION_KEY // 32 bytes
const IV_LENGTH = 16

function encrypt(text: string) {
  const iv = crypto.randomBytes(IV_LENGTH)
  const cipher = crypto.createCipheriv('aes-256-cbc', ENCRYPTION_KEY, iv)
  let encrypted = cipher.update(text)
  encrypted = Buffer.concat([encrypted, cipher.final()])
  return iv.toString('hex') + ':' + encrypted.toString('hex')
}

function decrypt(text: string) {
  const parts = text.split(':')
  const iv = Buffer.from(parts[0], 'hex')
  const encrypted = Buffer.from(parts[1], 'hex')
  const decipher = crypto.createDecipheriv('aes-256-cbc', ENCRYPTION_KEY, iv)
  let decrypted = decipher.update(encrypted)
  decrypted = Buffer.concat([decrypted, decipher.final()])
  return decrypted.toString()
}

- Frontend: Next.js 14 (App Router) + TypeScript + TailwindCSS
- Backend: Supabase (PostgreSQL + Auth + Realtime)
- 배포: Vercel (메인 + 꿀단지 별도)
